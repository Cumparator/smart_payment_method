// Command server — HTTP API, воркеры и сверщик в одном процессе.
package main

import (
	"context"
	"log/slog"
	"net/http"
	"os"

	"github.com/Cumparator/smart_payment_method/internal/api"
	"github.com/Cumparator/smart_payment_method/internal/config"
	"github.com/Cumparator/smart_payment_method/internal/store"
)

func main() {
	cfg, err := config.Load()
	if err != nil {
		slog.Error("config", "err", err)
		os.Exit(1)
	}

	level := slog.LevelInfo
	switch cfg.LogLevel {
	case "debug":
		level = slog.LevelDebug
	case "warn":
		level = slog.LevelWarn
	case "error":
		level = slog.LevelError
	}
	slog.SetDefault(slog.New(slog.NewTextHandler(os.Stdout, &slog.HandlerOptions{Level: level})))

	db, err := store.New(context.Background(), cfg.Database)
	if err != nil {
		slog.Error("db", "err", err)
		os.Exit(1)
	}
	defer db.Close()

	h := api.New(db)
	mux := http.NewServeMux()
	mux.HandleFunc("GET /healthz", func(w http.ResponseWriter, _ *http.Request) {
		w.WriteHeader(http.StatusOK)
	})
	mux.HandleFunc("POST /payments", h.CreatePayment)
	mux.HandleFunc("GET /payments/{id}", h.GetPayment)

	slog.Info("listening", "addr", cfg.Addr)
	if err := http.ListenAndServe(cfg.Addr, mux); err != nil {
		slog.Error("server", "err", err)
		os.Exit(1)
	}
}

package api

import (
	"encoding/json"
	"net/http"

	"github.com/Cumparator/smart_payment_method/internal/payment"
	"github.com/Cumparator/smart_payment_method/internal/store"
)

type Handler struct {
	Store *store.DB
}

func New(s *store.DB) *Handler { return &Handler{Store: s} }

type createReq struct {
	Amount   int64  `json:"amount"`
	Currency string `json:"currency"`
}

func (h *Handler) CreatePayment(w http.ResponseWriter, r *http.Request) {
	key := r.Header.Get("Idempotency-Key")
	if key == "" {
		http.Error(w, "Idempotency-Key required", http.StatusBadRequest)
		return
	}
	var req createReq
	if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
		http.Error(w, "bad body", http.StatusBadRequest)
		return
	}
	p := payment.Payment{
		ID:             newID(),
		IdempotencyKey: key,
		Amount:         req.Amount,
		Currency:       req.Currency,
		State:          payment.Received,
		NextRunAt:      timeNow(),
	}
	created, exists, err := h.Store.CreatePayment(r.Context(), p)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}
	if exists && created.Amount != req.Amount {
		http.Error(w, "idempotency key reused with different amount", http.StatusUnprocessableEntity)
		return
	}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(created)
}

func (h *Handler) GetPayment(w http.ResponseWriter, r *http.Request) {
	id := r.PathValue("id")
	p, err := h.Store.GetPayment(r.Context(), id)
	if err != nil {
		http.Error(w, "not found", http.StatusNotFound)
		return
	}
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(p)
}

// helpers
func newID() string { return generateUUID() }

package api

import (
	"crypto/rand"
	"encoding/hex"
	"time"
)

func generateUUID() string {
	b := make([]byte, 16)
	rand.Read(b)
	return hex.EncodeToString(b[:4]) + "-" + hex.EncodeToString(b[4:6]) + "-" + hex.EncodeToString(b[6:8]) + "-" + hex.EncodeToString(b[8:10]) + "-" + hex.EncodeToString(b[10:])
}

func timeNow() time.Time { return time.Now().UTC() }

// Package channel — единый контракт платёжного канала. Ничего не знает о БД
// и состояниях платежа, поэтому имитатор заменяется реальным провайдером.
package channel

import "context"

type ChargeOutcome string

const (
	Approved     ChargeOutcome = "APPROVED"      // деньги списаны → SUCCEEDED
	Declined     ChargeOutcome = "DECLINED"      // бизнес-отказ → FAILED
	ChannelError ChargeOutcome = "CHANNEL_ERROR" // техническая ошибка → можно в другой канал
	Unknown      ChargeOutcome = "UNKNOWN"       // таймаут/сеть → RECONCILING, переключаться нельзя
)

type StatusOutcome string

const (
	StatusApproved StatusOutcome = "APPROVED"
	StatusDeclined StatusOutcome = "DECLINED"
	StatusNotFound StatusOutcome = "NOT_FOUND"
)

type ChargeResult struct {
	Outcome ChargeOutcome
	Reason  string
}

type StatusResult struct {
	Outcome StatusOutcome
}

type CancelResult struct {
	Confirmed bool
}

// Tariff — комиссия канала: Fixed + Percent * сумма. Fixed в копейках.
type Tariff struct {
	Fixed   int64
	Percent float64
}

func (t Tariff) Fee(amount int64) float64 {
	return float64(t.Fixed) + t.Percent*float64(amount)
}

// PaymentChannel реализуют все каналы. key — ключ попытки payment_id:attempt_no.
type PaymentChannel interface {
	ID() string
	Charge(ctx context.Context, key string, amount int64, currency string) (ChargeResult, error)
	Status(ctx context.Context, key string) (StatusResult, error)
	Cancel(ctx context.Context, key string) (CancelResult, error)
}

// Package clock — источник времени. Доменная логика не вызывает time.Now()
// напрямую, чтобы эксперименты могли идти на виртуальных часах.
package clock

import (
	"sync"
	"time"
)

type Clock interface {
	Now() time.Time
}

type Real struct{}

func (Real) Now() time.Time { return time.Now() }

// Virtual — часы, которые двигает сам прогон эксперимента.
type Virtual struct {
	mu  sync.Mutex
	now time.Time
}

func NewVirtual(start time.Time) *Virtual {
	return &Virtual{now: start}
}

func (v *Virtual) Now() time.Time {
	v.mu.Lock()
	defer v.mu.Unlock()
	return v.now
}

func (v *Virtual) Advance(d time.Duration) {
	v.mu.Lock()
	defer v.mu.Unlock()
	v.now = v.now.Add(d)
}

package payment

import (
	"fmt"
	"time"
)

// Payment — платёж. Сумма хранится в копейках.
type Payment struct {
	ID             string
	IdempotencyKey string
	Amount         int64
	Currency       string
	State          State
	AttemptsCount  int
	NextRunAt      time.Time
	CreatedAt      time.Time
	UpdatedAt      time.Time
}

type AttemptState string

const (
	AttemptStarted      AttemptState = "STARTED"
	AttemptApproved     AttemptState = "APPROVED"
	AttemptDeclined     AttemptState = "DECLINED"
	AttemptChannelError AttemptState = "CHANNEL_ERROR"
	AttemptUnknown      AttemptState = "UNKNOWN"
	AttemptCancelled    AttemptState = "CANCELLED"
)

// IsLive — попытка не завершена: канал ещё может списать деньги.
// UNKNOWN тоже «живая»: исход неизвестен, пока его не выяснит сверщик.
// Список состояний должен совпадать с частичным индексом attempts_one_live_idx.
func (s AttemptState) IsLive() bool {
	return s == AttemptStarted || s == AttemptUnknown
}

// attemptTransitions — допустимые переходы состояния попытки.
var attemptTransitions = map[AttemptState][]AttemptState{
	AttemptStarted: {AttemptApproved, AttemptDeclined, AttemptChannelError, AttemptUnknown, AttemptCancelled},
	AttemptUnknown: {AttemptApproved, AttemptDeclined, AttemptCancelled},
}

// ValidateAttemptTransition проверяет переход попытки до CAS-обновления в БД.
func ValidateAttemptTransition(from, to AttemptState) error {
	for _, s := range attemptTransitions[from] {
		if s == to {
			return nil
		}
	}
	return fmt.Errorf("payment: недопустимый переход попытки %s → %s", from, to)
}

// MaxAttempts — не более 3 попыток на платёж.
const MaxAttempts = 3

// Attempt — одна попытка провести платёж через канал.
type Attempt struct {
	PaymentID  string
	AttemptNo  int
	ChannelID  string
	State      AttemptState
	StartedAt  time.Time
	FinishedAt *time.Time
}

// IsLive — см. AttemptState.IsLive.
func (a Attempt) IsLive() bool { return a.State.IsLive() }

// ChannelKey — ключ идемпотентности попытки, передаваемый в канал.
func (a Attempt) ChannelKey() string {
	return fmt.Sprintf("%s:%d", a.PaymentID, a.AttemptNo)
}

// Package payment — доменная модель платежа и автомат состояний.
// Это единственный пакет, которому разрешено менять статус платежа.
package payment

import "fmt"

type State string

const (
	Received     State = "RECEIVED"
	Processing   State = "PROCESSING"
	Reconciling  State = "RECONCILING"
	Succeeded    State = "SUCCEEDED"
	Failed       State = "FAILED"
	ManualReview State = "MANUAL_REVIEW"
)

// transitions — допустимые переходы. Processing → Received нужен для повтора
// в другой канал после CHANNEL_ERROR (см. wiki, «Открытые вопросы», п. 1).
var transitions = map[State][]State{
	Received:    {Processing},
	Processing:  {Succeeded, Failed, Received, Reconciling},
	Reconciling: {Succeeded, Failed, Received, ManualReview},
}

func (s State) IsFinal() bool {
	return s == Succeeded || s == Failed || s == ManualReview
}

func CanTransition(from, to State) bool {
	for _, s := range transitions[from] {
		if s == to {
			return true
		}
	}
	return false
}

// ValidateTransition проверяет переход до CAS-обновления в БД:
// UPDATE payments SET state = $to WHERE id = $id AND state = $from.
func ValidateTransition(from, to State) error {
	if !CanTransition(from, to) {
		return fmt.Errorf("payment: недопустимый переход %s → %s", from, to)
	}
	return nil
}

package payment

import "testing"

func TestTransitions(t *testing.T) {
	cases := []struct {
		from, to State
		ok       bool
	}{
		{Received, Processing, true},
		{Processing, Succeeded, true},
		{Processing, Reconciling, true},
		{Processing, Received, true},
		{Reconciling, ManualReview, true},
		{Reconciling, Received, true},
		{Received, Succeeded, false},
		{Succeeded, Received, false},
		{Failed, Processing, false},
		{ManualReview, Received, false},
	}
	for _, c := range cases {
		if got := CanTransition(c.from, c.to); got != c.ok {
			t.Errorf("CanTransition(%s, %s) = %v, want %v", c.from, c.to, got, c.ok)
		}
	}
}

func TestFinalStatesHaveNoTransitions(t *testing.T) {
	for _, s := range []State{Succeeded, Failed, ManualReview} {
		if !s.IsFinal() {
			t.Errorf("%s должен быть финальным", s)
		}
		if len(transitions[s]) != 0 {
			t.Errorf("из финального %s не должно быть переходов", s)
		}
	}
}

package simulator

import (
	"fmt"
	"time"
)

// DriftPoint — доля технических ошибок канала через At после Config.Start.
// Между соседними точками доля меняется линейно; две точки с одинаковым At
// дают скачок. До первой и после последней точки доля постоянна.
type DriftPoint struct {
	At        time.Duration
	ErrorRate float64
}

func validateDrift(c Config) error {
	for i, p := range c.Drift {
		if i > 0 && p.At < c.Drift[i-1].At {
			return fmt.Errorf("simulator %s: точки дрейфа не по порядку", c.ID)
		}
		if p.ErrorRate < 0 || p.ErrorRate > 1 {
			return fmt.Errorf("simulator %s: доля вне 0..1", c.ID)
		}
		if c.DeclineRate+p.ErrorRate+c.TimeoutRate > 1 {
			return fmt.Errorf("simulator %s: сумма долей больше 1", c.ID)
		}
	}
	return nil
}

// errorRateAt — доля технических ошибок в момент t.
func (c Config) errorRateAt(t time.Time) float64 {
	if len(c.Drift) == 0 {
		return c.ErrorRate
	}
	d := t.Sub(c.Start)
	if d < c.Drift[0].At {
		return c.Drift[0].ErrorRate
	}
	for i := len(c.Drift) - 1; i > 0; i-- {
		a, b := c.Drift[i-1], c.Drift[i]
		if d < a.At {
			continue
		}
		if d >= b.At {
			return b.ErrorRate
		}
		k := float64(d-a.At) / float64(b.At-a.At)
		return a.ErrorRate + k*(b.ErrorRate-a.ErrorRate)
	}
	return c.Drift[len(c.Drift)-1].ErrorRate
}

package simulator

import (
	"context"
	"sync"
	"time"

type EntryStatus string

const (
	Charged   EntryStatus = "CHARGED"
	Cancelled EntryStatus = "CANCELLED"
)

// Entry — запись книги списаний канала. Ключ — payment_id:attempt_no.
type Entry struct {
	ChannelID string
	Key       string
	Amount    int64
	Status    EntryStatus
	At        time.Time
}

// Ledger — книга списаний. Одна запись на (канал, ключ), поэтому повтор
// с тем же ключом не спишет деньги дважды.
type Ledger interface {
	Get(ctx context.Context, channelID, key string) (Entry, bool, error)
	// Charge добавляет запись, если её ещё нет, и возвращает актуальную.
	Charge(ctx context.Context, e Entry) (Entry, error)
	// Cancel переводит запись в CANCELLED; если записи нет — создаёт
	// отменённую с нулевой суммой, чтобы ключ больше не списывался.
	Cancel(ctx context.Context, channelID, key string, at time.Time) (Entry, error)
	Entries(ctx context.Context, channelID string) ([]Entry, error)
}

type ledgerKey struct{ channel, key string }

// MemLedger — книга в памяти для тестов и экспериментов.
type MemLedger struct {
	mu      sync.Mutex
	entries map[ledgerKey]Entry
	order   []ledgerKey
}

func NewMemLedger() *MemLedger {
	return &MemLedger{entries: make(map[ledgerKey]Entry)}
}

func (l *MemLedger) Get(_ context.Context, channelID, key string) (Entry, bool, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	e, ok := l.entries[ledgerKey{channelID, key}]
	return e, ok, nil
}

func (l *MemLedger) Charge(_ context.Context, e Entry) (Entry, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	k := ledgerKey{e.ChannelID, e.Key}
	if old, ok := l.entries[k]; ok {
		return old, nil
	}
	e.Status = Charged
	l.entries[k] = e
	l.order = append(l.order, k)
	return e, nil
}

func (l *MemLedger) Cancel(_ context.Context, channelID, key string, at time.Time) (Entry, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	k := ledgerKey{channelID, key}
	e, ok := l.entries[k]
	if !ok {
		e = Entry{ChannelID: channelID, Key: key}
		l.order = append(l.order, k)
	}
	if e.Status != Cancelled {
		e.Status = Cancelled
		e.At = at
	}
	l.entries[k] = e
	return e, nil
}

func (l *MemLedger) Entries(_ context.Context, channelID string) ([]Entry, error) {
	l.mu.Lock()
	defer l.mu.Unlock()
	var out []Entry
	for _, k := range l.order {
		if k.channel == channelID {
			out = append(out, l.entries[k])
		}
	}
	return out, nil
}

// Config — параметры имитатора. Доли задаются в 0..1, их сумма не больше 1;
// остаток — доля одобренных списаний.
type Config struct {
	ID          string
	Tariff      channel.Tariff
	Seed        int64
	DeclineRate float64 // бизнес-отказы → DECLINED
	ErrorRate   float64 // технические ошибки → CHANNEL_ERROR
	TimeoutRate float64 // потерянные ответы → UNKNOWN
	// TimeoutChargeRate — доля таймаутов, при которых деньги всё же списаны.
	TimeoutChargeRate float64

	Latency time.Duration // базовая задержка ответа
	Jitter  time.Duration // случайная добавка к задержке, 0..Jitter
	// Sleep ждёт d. По умолчанию — реальное ожидание; в экспериментах
	// подменяется сдвигом виртуальных часов.
	Sleep func(ctx context.Context, d time.Duration) error

	// Drift — изменение ErrorRate во времени; если задан, ErrorRate не используется.
	// Start — точка отсчёта, по умолчанию время создания канала по его часам.
	Drift []DriftPoint
	Start time.Time
}

func (c Config) validate() error {
	for _, r := range []float64{c.DeclineRate, c.ErrorRate, c.TimeoutRate, c.TimeoutChargeRate} {
		if r < 0 || r > 1 {
			return fmt.Errorf("simulator %s: доля вне 0..1", c.ID)
		}
	}
	if c.DeclineRate+c.ErrorRate+c.TimeoutRate > 1 {
		return fmt.Errorf("simulator %s: сумма долей больше 1", c.ID)
	}
	if c.Latency < 0 || c.Jitter < 0 {
		return fmt.Errorf("simulator %s: отрицательная задержка", c.ID)
	}
	return validateDrift(c)
}

func sleep(ctx context.Context, d time.Duration) error {
	t := time.NewTimer(d)
	defer t.Stop()
	select {
	case <-t.C:
		return nil
	case <-ctx.Done():
		return ctx.Err()
	}
}

// Channel — имитатор платёжного канала. Исход попытки детерминирован:
// он зависит только от seed, канала, ключа попытки и времени по часам канала.
type Channel struct {
	cfg    Config
	ledger Ledger
	clock  clock.Clock

	mu       sync.Mutex
	declined map[string]bool
}

var _ channel.PaymentChannel = (*Channel)(nil)

func New(cfg Config, ledger Ledger, clk clock.Clock) (*Channel, error) {
	if err := cfg.validate(); err != nil {
		return nil, err
	}
	if cfg.Sleep == nil {
		cfg.Sleep = sleep
	}
	if cfg.Start.IsZero() {
		cfg.Start = clk.Now()
	}
	return &Channel{cfg: cfg, ledger: ledger, clock: clk, declined: make(map[string]bool)}, nil
}

func (c *Channel) ID() string { return c.cfg.ID }

func (c *Channel) Tariff() channel.Tariff { return c.cfg.Tariff }

// Latency — типичная задержка ответа канала.
func (c *Channel) Latency() time.Duration { return c.cfg.Latency + c.cfg.Jitter/2 }

// ErrorRateAt — истинная доля технических ошибок в момент t. Стратегиям
// недоступна; нужна эксперименту для сравнения с «оракулом».
func (c *Channel) ErrorRateAt(t time.Time) float64 { return c.cfg.errorRateAt(t) }

func (c *Channel) Charge(ctx context.Context, key string, amount int64, _ string) (channel.ChargeResult, error) {
	res, err := c.process(ctx, key, amount)
	if err != nil {
		return res, err
	}
	if err := c.wait(ctx, key); err != nil {
		// запрос канал уже обработал, но ответ до клиента не дошёл
		return channel.ChargeResult{Outcome: channel.Unknown, Reason: err.Error()}, nil
	}
	return res, nil
}

func (c *Channel) wait(ctx context.Context, key string) error {
	d := c.cfg.Latency + time.Duration(c.roll(key, "latency")*float64(c.cfg.Jitter))
	if d <= 0 {
		return ctx.Err()
	}
	return c.cfg.Sleep(ctx, d)
}

func (c *Channel) process(ctx context.Context, key string, amount int64) (channel.ChargeResult, error) {
	if res, ok, err := c.repeat(ctx, key); err != nil || ok {
		return res, err
	}

	r := c.roll(key, "outcome")
	errRate := c.cfg.errorRateAt(c.clock.Now())
	switch {
	case r < c.cfg.TimeoutRate:
		if c.roll(key, "timeout") < c.cfg.TimeoutChargeRate {
			if _, err := c.charge(ctx, key, amount); err != nil {
				return channel.ChargeResult{}, err
			}
		}
		return channel.ChargeResult{Outcome: channel.Unknown, Reason: "timeout"}, nil
	case r < c.cfg.TimeoutRate+errRate:
		return channel.ChargeResult{Outcome: channel.ChannelError, Reason: "provider error"}, nil
	case r < c.cfg.TimeoutRate+errRate+c.cfg.DeclineRate:
		c.mu.Lock()
		c.declined[key] = true
		c.mu.Unlock()
		return channel.ChargeResult{Outcome: channel.Declined, Reason: "insufficient funds"}, nil
	}

	e, err := c.charge(ctx, key, amount)
	if err != nil {
		return channel.ChargeResult{}, err
	}
	if e.Status == Cancelled {
		return channel.ChargeResult{Outcome: channel.Declined, Reason: "cancelled"}, nil
	}
	return channel.ChargeResult{Outcome: channel.Approved}, nil
}

// repeat отвечает на повтор по уже обработанному ключу, не списывая заново.
func (c *Channel) repeat(ctx context.Context, key string) (channel.ChargeResult, bool, error) {
	c.mu.Lock()
	declined := c.declined[key]
	c.mu.Unlock()
	if declined {
		return channel.ChargeResult{Outcome: channel.Declined, Reason: "insufficient funds"}, true, nil
	}
	e, ok, err := c.ledger.Get(ctx, c.cfg.ID, key)
	if err != nil || !ok {
		return channel.ChargeResult{}, false, err
	}
	if e.Status == Cancelled {
		return channel.ChargeResult{Outcome: channel.Declined, Reason: "cancelled"}, true, nil
	}
	return channel.ChargeResult{Outcome: channel.Approved}, true, nil
}

func (c *Channel) charge(ctx context.Context, key string, amount int64) (Entry, error) {
	return c.ledger.Charge(ctx, Entry{ChannelID: c.cfg.ID, Key: key, Amount: amount, At: c.clock.Now()})
}

func (c *Channel) Status(ctx context.Context, key string) (channel.StatusResult, error) {
	c.mu.Lock()
	declined := c.declined[key]
	c.mu.Unlock()
	if declined {
		return channel.StatusResult{Outcome: channel.StatusDeclined}, nil
	}
	e, ok, err := c.ledger.Get(ctx, c.cfg.ID, key)
	if err != nil {
		return channel.StatusResult{}, err
	}
	switch {
	case !ok:
		return channel.StatusResult{Outcome: channel.StatusNotFound}, nil
	case e.Status == Cancelled:
		return channel.StatusResult{Outcome: channel.StatusDeclined}, nil
	}
	return channel.StatusResult{Outcome: channel.StatusApproved}, nil
}

func (c *Channel) Cancel(ctx context.Context, key string) (channel.CancelResult, error) {
	if _, err := c.ledger.Cancel(ctx, c.cfg.ID, key, c.clock.Now()); err != nil {
		return channel.CancelResult{}, err
	}
	return channel.CancelResult{Confirmed: true}, nil
}

// Summary — итог по книге: сколько списано и сколько стоила комиссия.
type Summary struct {
	Charges   int
	Amount    int64
	Fees      float64
	Cancelled int
}

func (c *Channel) Summary(ctx context.Context) (Summary, error) {
	entries, err := c.ledger.Entries(ctx, c.cfg.ID)
	if err != nil {
		return Summary{}, err
	}
	var s Summary
	for _, e := range entries {
		if e.Status == Cancelled {
			s.Cancelled++
			continue
		}
		s.Charges++
		s.Amount += e.Amount
		s.Fees += c.cfg.Tariff.Fee(e.Amount)
	}
	return s, nil
}

// roll — детерминированное число в [0, 1) от hash(seed, channel, key, salt).
func (c *Channel) roll(key, salt string) float64 {
	h := fnv.New64a()
	fmt.Fprintf(h, "%d|%s|%s|%s", c.cfg.Seed, c.cfg.ID, key, salt)
	return float64(h.Sum64()>>11) / (1 << 53)
}

package store

import (
	"context"
	"errors"
	"fmt"
	"time"

	"github.com/Cumparator/smart_payment_method/internal/payment"
	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgconn"
)

var (
	// ErrLiveAttemptExists — у платежа уже есть незавершённая (STARTED/UNKNOWN) попытка.
	ErrLiveAttemptExists = errors.New("store: у платежа уже есть незавершённая попытка")
	// ErrMaxAttempts — исчерпан лимит payment.MaxAttempts.
	ErrMaxAttempts = errors.New("store: исчерпан лимит попыток платежа")
	// ErrPaymentNotFound — платежа с таким id нет.
	ErrPaymentNotFound = errors.New("store: платёж не найден")
)

const (
	pgUniqueViolation = "23505"
	// Частичный уникальный индекс из migrations/0001_init.sql:
	// UNIQUE (payment_id) WHERE state IN ('STARTED', 'UNKNOWN').
	liveAttemptIndex = "attempts_one_live_idx"
)

const attemptColumns = `payment_id, attempt_no, channel_id, state, started_at, finished_at`

func scanAttempt(row pgx.Row) (payment.Attempt, error) {
	var a payment.Attempt
	var state string
	var finished *time.Time
	if err := row.Scan(&a.PaymentID, &a.AttemptNo, &a.ChannelID, &state, &a.StartedAt, &finished); err != nil {
		return payment.Attempt{}, err
	}
	a.State = payment.AttemptState(state)
	a.FinishedAt = finished
	return a, nil
}

// StartAttempt создаёт новую попытку в состоянии STARTED. Вызывается ДО обращения к каналу.
//
// В одной транзакции: атомарно занимает следующий номер попытки (и блокирует строку
// платежа от конкурентных стартов) и вставляет строку в attempts. Одну живую попытку
// на платёж гарантирует частичный уникальный индекс, а не проверка в коде, поэтому
// гонка двух воркеров невозможна: проигравший получит ErrLiveAttemptExists,
// а его транзакция (вместе с увеличением attempts_count) откатится.
func (d *DB) StartAttempt(ctx context.Context, paymentID, channelID string) (payment.Attempt, error) {
	var att payment.Attempt
	err := d.Tx(ctx, func(tx pgx.Tx) error {
		var no int
		err := tx.QueryRow(ctx, `
			UPDATE payments SET attempts_count = attempts_count + 1, updated_at = now()
			WHERE id = $1 AND attempts_count < $2
			RETURNING attempts_count`, paymentID, payment.MaxAttempts).Scan(&no)
		if errors.Is(err, pgx.ErrNoRows) {
			return classifyStartRefusal(ctx, tx, paymentID)
		}
		if err != nil {
			return err
		}

		a := payment.Attempt{PaymentID: paymentID, AttemptNo: no, ChannelID: channelID, State: payment.AttemptStarted}
		att, err = scanAttempt(tx.QueryRow(ctx, `
			INSERT INTO attempts (payment_id, attempt_no, channel_id, channel_key, state)
			VALUES ($1, $2, $3, $4, $5)
			RETURNING `+attemptColumns,
			a.PaymentID, a.AttemptNo, a.ChannelID, a.ChannelKey(), string(a.State)))
		var pgErr *pgconn.PgError
		if errors.As(err, &pgErr) && pgErr.Code == pgUniqueViolation && pgErr.ConstraintName == liveAttemptIndex {
			return ErrLiveAttemptExists
		}
		return err
	})
	if err != nil {
		return payment.Attempt{}, err
	}
	return att, nil
}

// classifyStartRefusal объясняет, почему не удалось занять номер попытки.
func classifyStartRefusal(ctx context.Context, tx pgx.Tx, paymentID string) error {
	var count int
	var live bool
	err := tx.QueryRow(ctx, `
		SELECT p.attempts_count,
		       EXISTS (SELECT 1 FROM attempts a
		               WHERE a.payment_id = p.id AND a.state IN ('STARTED', 'UNKNOWN'))
		FROM payments p WHERE p.id = $1`, paymentID).Scan(&count, &live)
	if errors.Is(err, pgx.ErrNoRows) {
		return ErrPaymentNotFound
	}
	if err != nil {
		return err
	}
	if live {
		return ErrLiveAttemptExists
	}
	return fmt.Errorf("%w: %d из %d", ErrMaxAttempts, count, payment.MaxAttempts)
}

// SetAttemptState переводит попытку from → to (CAS по текущему состоянию).
// Возвращает false, если попытка уже не в состоянии from (кто-то успел раньше).
// finished_at ставится только при переходе в завершённое состояние;
// UNKNOWN остаётся «живым», поэтому finished_at у него пустой.
func (d *DB) SetAttemptState(ctx context.Context, paymentID string, attemptNo int, from, to payment.AttemptState) (bool, error) {
	if err := payment.ValidateAttemptTransition(from, to); err != nil {
		return false, err
	}
	var finishedAt *time.Time
	if !to.IsLive() {
		now := time.Now()
		finishedAt = &now
	}
	tag, err := d.Pool.Exec(ctx, `
		UPDATE attempts SET state = $1, finished_at = $2
		WHERE payment_id = $3 AND attempt_no = $4 AND state = $5`,
		string(to), finishedAt, paymentID, attemptNo, string(from))
	if err != nil {
		return false, err
	}
	return tag.RowsAffected() == 1, nil
}

// GetLiveAttempt возвращает незавершённую попытку платежа, если она есть.
func (d *DB) GetLiveAttempt(ctx context.Context, paymentID string) (payment.Attempt, bool, error) {
	a, err := scanAttempt(d.Pool.QueryRow(ctx, `
		SELECT `+attemptColumns+` FROM attempts
		WHERE payment_id = $1 AND state IN ('STARTED', 'UNKNOWN')`, paymentID))
	if errors.Is(err, pgx.ErrNoRows) {
		return payment.Attempt{}, false, nil
	}
	if err != nil {
		return payment.Attempt{}, false, err
	}
	return a, true, nil
}

// ListAttempts возвращает все попытки платежа по возрастанию номера.
func (d *DB) ListAttempts(ctx context.Context, paymentID string) ([]payment.Attempt, error) {
	rows, err := d.Pool.Query(ctx, `
		SELECT `+attemptColumns+` FROM attempts WHERE payment_id = $1 ORDER BY attempt_no`, paymentID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []payment.Attempt
	for rows.Next() {
		a, err := scanAttempt(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, a)
	}
	return out, rows.Err()
}

// Ledger — книга списаний имитатора в таблице channel_ledger.
type Ledger struct {
	db *DB
}

var _ simulator.Ledger = (*Ledger)(nil)

func NewLedger(db *DB) *Ledger { return &Ledger{db: db} }

const ledgerCols = `channel_id, channel_key, amount, status, at`

func scanEntry(row pgx.Row) (simulator.Entry, error) {
	var e simulator.Entry
	err := row.Scan(&e.ChannelID, &e.Key, &e.Amount, &e.Status, &e.At)
	return e, err
}

// Get loads entry by channel and key.
func (l *Ledger) Get(ctx context.Context, channelID, key string) (simulator.Entry, bool, error) {
	e, err := scanEntry(l.db.Pool.QueryRow(ctx,
		`SELECT `+ledgerCols+` FROM channel_ledger WHERE channel_id=$1 AND channel_key=$2`, channelID, key))
	if err == pgx.ErrNoRows {
		return simulator.Entry{}, false, nil
	}
	if err != nil {
		return simulator.Entry{}, false, err
	}
	return e, true, nil
}

// Charge inserts CHARGED entry; returns existing if key already used.
func (l *Ledger) Charge(ctx context.Context, e simulator.Entry) (simulator.Entry, error) {
	got, err := scanEntry(l.db.Pool.QueryRow(ctx, `
		INSERT INTO channel_ledger (channel_id, channel_key, amount, status, at)
		VALUES ($1,$2,$3,$4,$5)
		ON CONFLICT (channel_id, channel_key) DO NOTHING
		RETURNING `+ledgerCols,
		e.ChannelID, e.Key, e.Amount, string(simulator.Charged), e.At))
	if err == pgx.ErrNoRows {
		got, _, err = l.Get(ctx, e.ChannelID, e.Key)
	}
	return got, err
}

// Cancel marks entry CANCELLED, creating an empty one if absent.
func (l *Ledger) Cancel(ctx context.Context, channelID, key string, at time.Time) (simulator.Entry, error) {
	got, err := scanEntry(l.db.Pool.QueryRow(ctx, `
		INSERT INTO channel_ledger (channel_id, channel_key, amount, status, at)
		VALUES ($1,$2,0,$3,$4)
		ON CONFLICT (channel_id, channel_key) DO UPDATE SET status=EXCLUDED.status, at=EXCLUDED.at
		WHERE channel_ledger.status <> EXCLUDED.status
		RETURNING `+ledgerCols,
		channelID, key, string(simulator.Cancelled), at))
	if err == pgx.ErrNoRows {
		got, _, err = l.Get(ctx, channelID, key)
	}
	return got, err
}

// Entries lists channel entries in insertion order.
func (l *Ledger) Entries(ctx context.Context, channelID string) ([]simulator.Entry, error) {
	rows, err := l.db.Pool.Query(ctx,
		`SELECT `+ledgerCols+` FROM channel_ledger WHERE channel_id=$1 ORDER BY at, channel_key`, channelID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()
	var out []simulator.Entry
	for rows.Next() {
		e, err := scanEntry(rows)
		if err != nil {
			return nil, err
		}
		out = append(out, e)
	}
	return out, rows.Err()
}

// PaymentRow — строка payments.
type PaymentRow struct {
	ID             string
	IdempotencyKey string
	Amount         int64
	Currency       string
	State          payment.State
	AttemptsCount  int
	NextRunAt      time.Time
	CreatedAt      time.Time
	UpdatedAt      time.Time

func toPayment(r PaymentRow) payment.Payment {
	return payment.Payment{
		ID: r.ID, IdempotencyKey: r.IdempotencyKey, Amount: r.Amount,
		Currency: r.Currency, State: r.State, AttemptsCount: r.AttemptsCount,
		NextRunAt: r.NextRunAt, CreatedAt: r.CreatedAt, UpdatedAt: r.UpdatedAt,
	}
}

// CreatePayment inserts payment; returns existing if idempotency key exists.
func (d *DB) CreatePayment(ctx context.Context, p payment.Payment) (payment.Payment, bool, error) {
	var row PaymentRow
	err := d.Pool.QueryRow(ctx, `
		INSERT INTO payments (id, idempotency_key, amount, currency, state, attempts_count, next_run_at)
		VALUES ($1,$2,$3,$4,$5,$6,$7)
		ON CONFLICT (idempotency_key) DO NOTHING
		RETURNING id, idempotency_key, amount, currency, state, attempts_count, next_run_at, created_at, updated_at`,
		p.ID, p.IdempotencyKey, p.Amount, p.Currency, string(p.State), p.AttemptsCount, p.NextRunAt,
	).Scan(&row.ID, &row.IdempotencyKey, &row.Amount, &row.Currency, &row.State, &row.AttemptsCount, &row.NextRunAt, &row.CreatedAt, &row.UpdatedAt)
	if err == pgx.ErrNoRows {
		// conflict: load existing
		err2 := d.Pool.QueryRow(ctx, `
			SELECT id, idempotency_key, amount, currency, state, attempts_count, next_run_at, created_at, updated_at
			FROM payments WHERE idempotency_key=$1`, p.IdempotencyKey).Scan(
			&row.ID, &row.IdempotencyKey, &row.Amount, &row.Currency, &row.State, &row.AttemptsCount, &row.NextRunAt, &row.CreatedAt, &row.UpdatedAt)
		if err2 != nil {
			return payment.Payment{}, false, err2
		}
		return toPayment(row), true, nil
	}
	if err != nil {
		return payment.Payment{}, false, err
	}
	return toPayment(row), false, nil
}

// GetPayment loads payment by id.
func (d *DB) GetPayment(ctx context.Context, id string) (payment.Payment, error) {
	var row PaymentRow
	err := d.Pool.QueryRow(ctx, `
		SELECT id, idempotency_key, amount, currency, state, attempts_count, next_run_at, created_at, updated_at
		FROM payments WHERE id=$1`, id).Scan(
		&row.ID, &row.IdempotencyKey, &row.Amount, &row.Currency, &row.State, &row.AttemptsCount, &row.NextRunAt, &row.CreatedAt, &row.UpdatedAt)
	if err != nil {
		return payment.Payment{}, err
	}
	return toPayment(row), nil
}

// TransitionStatus updates state if old matches, appends event.
func (d *DB) TransitionStatus(ctx context.Context, id string, from, to payment.State, reason string) (bool, error) {
	var ok bool
	err := d.Tx(ctx, func(tx pgx.Tx) error {
		tag, err := tx.Exec(ctx, `UPDATE payments SET state=$1, updated_at=now() WHERE id=$2 AND state=$3`, string(to), id, string(from))
		if err != nil {
			return err
		}
		if tag.RowsAffected() == 0 {
			ok = false
			return nil
		}
		ok = true
		_, err = tx.Exec(ctx, `INSERT INTO payment_events (payment_id, from_state, to_state, reason) VALUES ($1,$2,$3,$4)`, id, string(from), string(to), reason)
		return err
	})
	return ok, err
}

package store

import (

	"github.com/jackc/pgx/v5"
	"github.com/jackc/pgx/v5/pgxpool"
)

// DB — пул соединений.
type DB struct {
	Pool *pgxpool.Pool
}

// New открывает пул.
func New(ctx context.Context, dsn string) (*DB, error) {
	pool, err := pgxpool.New(ctx, dsn)
	if err != nil {
		return nil, fmt.Errorf("pgxpool.New: %w", err)
	}
	if err := pool.Ping(ctx); err != nil {
		pool.Close()
		return nil, fmt.Errorf("ping: %w", err)
	}
	return &DB{Pool: pool}, nil
}

// Close закрывает пул.
func (d *DB) Close() { d.Pool.Close() }

// Tx выполняет fn в транзакции.
func (d *DB) Tx(ctx context.Context, fn func(pgx.Tx) error) error {
	tx, err := d.Pool.Begin(ctx)
	if err != nil {
		return err
	}
	defer tx.Rollback(ctx)
	if err := fn(tx); err != nil {
		return err
	}
	return tx.Commit(ctx)
}

package strategy

// ExpectedCost — Стратегия 2:
// Цена = Комиссия + (1 − p_success)·Штраф + Задержка·ЦенаСекунды (+ Добавка за загрузку).
// Все денежные параметры — в копейках.
type ExpectedCost struct {
	FailurePenalty float64 // условная стоимость неудачной попытки
	SecondCost     float64 // стоимость секунды ожидания клиента
	LoadThreshold  float64 // порог загрузки, после которого начисляется добавка (0.7)
	LoadPenalty    float64 // добавка за каждую долю загрузки сверх порога
}

func (ExpectedCost) Name() string { return "expected_cost" }

func (e ExpectedCost) Score(amount int64, candidates []Candidate) []Scored {
	out := make([]Scored, len(candidates))
	for i, c := range candidates {
		score := c.Tariff.Fee(amount) +
			(1-c.PSuccess())*e.FailurePenalty +
			c.LatencyS*e.SecondCost
		if c.Load > e.LoadThreshold {
			score += (c.Load - e.LoadThreshold) * e.LoadPenalty
		}
		out[i] = Scored{Candidate: c, Score: score}
	}
	return out
}

// Package strategy — оценка допустимых каналов. Стратегии — чистые функции
// от снимка статистики, поэтому одинаково работают в живом режиме и в эксперименте.
package strategy

import (
	"sort"

	"github.com/Cumparator/smart_payment_method/internal/channel"
)

// Candidate — снимок канала, прошедшего фильтрацию маршрутизатора.
type Candidate struct {
	ChannelID string
	Tariff    channel.Tariff
	Successes int     // технические успехи в скользящем окне
	Attempts  int     // технические исходы в окне (DECLINED не считается)
	LatencyS  float64 // типичная задержка ответа, секунды
	Load      float64 // загрузка канала, 0..1
}

// PSuccess — вероятность успеха со сглаживанием Лапласа: (S + 1) / (N + 2).
func (c Candidate) PSuccess() float64 {
	return float64(c.Successes+1) / float64(c.Attempts+2)
}

type Scored struct {
	Candidate
	Score float64 // меньше — лучше
}

type Strategy interface {
	Name() string
	Score(amount int64, candidates []Candidate) []Scored
}

// Rank оценивает кандидатов и сортирует их по возрастанию оценки.
func Rank(s Strategy, amount int64, candidates []Candidate) []Scored {
	scored := s.Score(amount, candidates)
	sort.SliceStable(scored, func(i, j int) bool { return scored[i].Score < scored[j].Score })
	return scored
}

# Smart Payment Method

Сервис умной маршрутизации платежей. Для каждого платежа он выбирает лучший из нескольких платёжных каналов с учётом цены, надёжности и задержек и при этом гарантирует отсутствие двойных списаний.

Проект исследовательский: две стратегии выбора канала сравниваются в воспроизводимых экспериментах, результаты идут в научную статью.

> 📚 Подробная документация — в [wiki](https://github.com/Cumparator/smart_payment_method/wiki), начать стоит со страницы [«Базовая архитектура»](https://github.com/Cumparator/smart_payment_method/wiki/Базовая-архитектура).

## Главные идеи

- **Таймаут — это не отказ.** Если ответ канала неясен (`UNKNOWN`), платёж уходит в сверку (`RECONCILING`), а не в повтор через другой канал.
- **Защита от двойного списания в несколько слоёв:** `Idempotency-Key` на входе, не больше одной «живой» попытки на платёж (частичный уникальный индекс), ключ попытки `payment_id:attempt_no` в канале, попытка записывается в БД *до* вызова канала.
- **Устойчивость:** у каждого канала свой circuit breaker, повторы идут с экспоненциальной паузой и jitter, общий бюджет повторов — 20% потока.
- **Объяснимость:** каждое решение маршрутизатора сохраняется вместе с причинами, и панель показывает его обычной фразой.
- **Воспроизводимость:** исходы каналов детерминированы через `hash(seed, channel, payment, attempt)`, эксперименты идут на виртуальных часах, каждый прогоняется на 30 seed'ах.

## Архитектура

Модульный монолит на Go: HTTP API, фоновые воркеры и сверщик работают в одном процессе. PostgreSQL служит и хранилищем, и очередью задач (`SELECT … FOR UPDATE SKIP LOCKED`). Платёжные каналы — имитаторы за единым интерфейсом `PaymentChannel`. Панель управления на React.

```text
Клиент ──POST /payments──▶ API ──▶ PostgreSQL ◀── воркеры ──▶ маршрутизатор ──▶ стратегия
                                       ▲              │
                                       │              ▼
                                   сверщик ──────▶ каналы (имитаторы)
```

Две стратегии:
1. **Самый дешёвый** — минимальная комиссия, контрольная группа.
2. **Ожидаемая цена** — `комиссия + (1 − p_success)·штраф + задержка·цена_секунды (+ добавка за загрузку)`.

## Структура репозитория

  scenariogen/   генератор сценариев экспериментов (.jsonl)

# React + TypeScript + Vite

Currently, two official plugins are available:

- [@vitejs/plugin-react](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react) uses [Oxc](https://oxc.rs)
- [@vitejs/plugin-react-swc](https://github.com/vitejs/vite-plugin-react/blob/main/packages/plugin-react-swc) uses [SWC](https://swc.rs/)

## React Compiler

The React Compiler is not enabled on this template because of its impact on dev & build performances. To add it, see [this documentation](https://react.dev/learn/react-compiler/installation).

export default defineConfig([
